cmake_minimum_required(VERSION 3.24)
project(ashfall VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ASHFALL_BRIDGE "Build the Python-compatible C ABI" ON)
option(ASHFALL_VIEWER "Build raylib viewer" OFF)
option(ASHFALL_STATISTICAL_TESTS "Validate recorded forest samples and fit numerics" OFF)
option(ASHFALL_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)
include(FetchContent)
FetchContent_Declare(tomlplusplus URL https://codeload.github.com/marzer/tomlplusplus/tar.gz/refs/tags/v3.4.0 URL_HASH SHA256=8517f65938a4faae9ccf8ebb36631a38c1cadfb5efa85d9a72e15b9e97d25155 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(xxhash URL https://codeload.github.com/Cyan4973/xxHash/tar.gz/refs/tags/v0.8.3 URL_HASH SHA256=aae608dfe8213dfd05d909a57718ef82f30722c392344583d3f39050c7f29a80 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(pcg URL https://codeload.github.com/imneme/pcg-cpp/tar.gz/428802d1a5634f96bcd0705fab379ff0113bcf13 URL_HASH SHA256=79f23706ed0cbc1bb57ea35d50d0abed66898b8f7b0bcf2aaecdfd234863b060 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(tomlplusplus xxhash pcg)
find_package(Threads REQUIRED)
add_library(ashfall_options INTERFACE)
if(MSVC)
  target_compile_options(ashfall_options INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(ashfall_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wold-style-cast -Werror)
  if(ASHFALL_SANITIZERS)
    target_compile_options(ashfall_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ashfall_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(ashfall_hash STATIC ${xxhash_SOURCE_DIR}/xxhash.c)
target_include_directories(ashfall_hash SYSTEM PUBLIC ${xxhash_SOURCE_DIR})
add_library(ashfall_core STATIC src/core/config.cpp)
target_include_directories(ashfall_core PUBLIC include)
target_include_directories(ashfall_core SYSTEM PUBLIC ${pcg_SOURCE_DIR}/include)
target_link_libraries(ashfall_core PUBLIC ashfall_options ashfall_hash tomlplusplus::tomlplusplus Threads::Threads)
target_compile_definitions(ashfall_core PUBLIC TOML_EXCEPTIONS=0)
add_library(ashfall_sim STATIC src/sim/world.cpp src/sim/terrain.cpp src/sim/forest.cpp src/sim/kingdom.cpp src/agent/action.cpp)
target_link_libraries(ashfall_sim PUBLIC ashfall_core)
if(NOT MSVC)
  target_compile_options(ashfall_core PRIVATE -fno-exceptions -fno-rtti)
  target_compile_options(ashfall_sim PRIVATE -fno-exceptions -fno-rtti)
endif()
add_executable(ashfall_headless src/tools/headless_main.cpp)
target_link_libraries(ashfall_headless PRIVATE ashfall_sim)
add_executable(ashfall_analyze src/tools/analyze_main.cpp)
target_link_libraries(ashfall_analyze PRIVATE ashfall_core)
add_executable(ashfall_replay src/tools/replay_main.cpp)
target_link_libraries(ashfall_replay PRIVATE ashfall_sim)
if(ASHFALL_BRIDGE)
  add_library(ashfall_bridge SHARED src/bridge/c_api.cpp)
  target_link_libraries(ashfall_bridge PRIVATE ashfall_sim)
endif()
if(BUILD_TESTING)
  FetchContent_Declare(doctest URL https://codeload.github.com/doctest/doctest/tar.gz/refs/tags/v2.4.11 URL_HASH SHA256=632ed2c05a7f53fa961381497bf8069093f0d6628c5f26286161fbd32a560186 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(doctest)
  add_executable(ashfall_tests tests/core_tests.cpp tests/allocation_tests.cpp)
  target_link_libraries(ashfall_tests PRIVATE ashfall_sim doctest::doctest)
  add_test(NAME unit_property COMMAND ashfall_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME analysis COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/analysis_tests.py)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py --bin $<TARGET_FILE_DIR:ashfall_headless>)
  if(ASHFALL_STATISTICAL_TESTS)
    add_test(NAME forest_measurement COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/python/analysis/forest.py ${CMAKE_CURRENT_SOURCE_DIR}/docs/data/fires-128-seed1.csv.gz ${CMAKE_CURRENT_SOURCE_DIR}/docs/data/fires-256-seed1.csv.gz --check)
    set_tests_properties(forest_measurement PROPERTIES LABELS statistical)
  endif()
  add_test(NAME empty_10000 COMMAND ashfall_headless --empty --ticks 10000)
endif()
if(ASHFALL_VIEWER)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5 URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(raylib)
  add_executable(ashfall_viewer src/render/viewer.cpp)
  target_link_libraries(ashfall_viewer PRIVATE ashfall_sim raylib)
endif()
